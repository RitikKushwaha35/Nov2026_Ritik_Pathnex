stages:

node-install:
  stage: install
  image: node:16
  script:

    - npm install

  stage: test
  image: node:16
  script:
    - cd sample-node-app
    - npm test

node-build:
  stage: build
  image: node:16
  script:
    - cd sample-node-app
    - npm run build
